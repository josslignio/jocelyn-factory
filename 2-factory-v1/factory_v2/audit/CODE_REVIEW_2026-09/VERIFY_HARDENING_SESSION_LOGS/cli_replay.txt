import json, os, subprocess, sys, tempfile
from pathlib import Path
root=Path.cwd(); results=[]
def run(name, cmd, cwd=root, expected=0, contains=''):
    p=subprocess.run(cmd,cwd=cwd,text=True,capture_output=True,env={**os.environ,'FACTORY_NO_REAL_ENGINE':'1'})
    assert p.returncode==expected,(name,p.returncode,p.stdout,p.stderr)
    output=p.stdout+p.stderr
    assert contains in output,(name,output)
    results.append({'case':name,'command':cmd,'cwd':str(cwd),'rc':p.returncode,'output':output.strip()})
    print(f'TEST {name} : rc={p.returncode} ; {output.strip()[:180]}')
